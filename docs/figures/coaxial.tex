% Corona inception on the inner conductor of a coaxial arrangement.
%
% Data files (in the working directory when this is compiled):
%   sim_a0.25mm.dat, sim_a1mm.dat, sim_a4mm.dat
%       incept1d pdiv output from docs/figures/Makefile
%       (examples/coaxial/README.md), outer radius b = 10 mm
%
% 0-based column indices (\thisrowno) into sim_*.dat; 1-based layout:
%    1  p_bar
%    4  EN_Td[... (inner=positive)]     9  EN_Td[... (inner=negative)]
%
% EN_Td is the mean reduced field U/(N d).  The figure plots the reduced field
% at the inner conductor instead,
%   E(a)/N = f(0) * EN_Td,   f(0) = (b - a) / (a ln(b/a)),
% against p*a, the similarity variables of the drift-reaction model near
% the wire.
\documentclass[10pt]{article}
\input{preamble.tex}
\usepackage[paperwidth=15cm,paperheight=9.4cm,margin=0in]{geometry}

\pgfplotsset{
  pos/.style={very thick, no marks},
  neg/.style={very thick, no marks, densely dashed},
}
\setlength{\figwidth}{14cm}
\setlength{\figheight}{9cm}

% Outer radius in mm.
\def\bmm{10}

% \coax{file}{a_mm}{column}{style}{colour}
\newcommand{\coax}[5]{%
  \addplot[#4, #5] table[
    x expr={\thisrowno{0}*#2},
    y expr={\thisrowno{#3}*(\bmm - #2)/(#2*ln(\bmm/#2))},
  ] {#1};
}

\definecolor{c1}{RGB}{228,26,28}
\definecolor{c2}{RGB}{55,126,184}
\definecolor{c3}{RGB}{77,175,74}

\begin{document}
\centering
\begin{tikzpicture}
  \begin{axis}[
      anchor=south west,
      axis y line=left,
      axis x line=bottom,
      font=\small,
      width=\figwidth,
      height=\figheight,
      grid=both,
      minor grid style={very thin, densely dotted},
      axis line style={thick, black},
      unbounded coords=jump,
      xmode=log, ymode=log,
      xmin=2e-2, xmax=50,
      ymin=100, ymax=1000,
      ytick={100,200,300,500,1000},
      yticklabels={100,200,300,500,1000},
      ylabel={$E(a)/N$ (\si{\townsend})},
      xlabel={$pa$ (\si{\bar\milli\meter})},
      boxedlegend,
      legend style={at={(0.97, 0.97)}, anchor=north east},
    ]

    \coax{sim_a0.25mm.dat}{0.25}{3}{pos}{c1}
    \addlegendentry{$a = \SI{0.25}{\milli\meter}$};
    \coax{sim_a1mm.dat}{1}{3}{pos}{c2}
    \addlegendentry{$a = \SI{1}{\milli\meter}$};
    \coax{sim_a4mm.dat}{4}{3}{pos}{c3}
    \addlegendentry{$a = \SI{4}{\milli\meter}$};

    \coax{sim_a0.25mm.dat}{0.25}{8}{neg, forget plot}{c1}
    \coax{sim_a1mm.dat}{1}{8}{neg, forget plot}{c2}
    \coax{sim_a4mm.dat}{4}{8}{neg, forget plot}{c3}

    % Polarity key.
    \addplot[pos, black] coordinates {(1e-9,1e-9)};
    \addlegendentry{inner conductor $+$};
    \addplot[neg, black] coordinates {(1e-9,1e-9)};
    \addlegendentry{inner conductor $-$};
  \end{axis}
\end{tikzpicture}
\end{document}
