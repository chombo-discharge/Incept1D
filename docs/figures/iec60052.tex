% Breakdown voltage of standard sphere gaps: IEC 60052 tables vs. Incept1D.
%
% Data files (all in the working directory when this is compiled):
%   sim_<D>cm.dat        incept1d pdiv output from docs/figures/Makefile (examples/iec60052/README.md)
%   iec60052_<D>cm.dat   OPTIONAL reference tables (pd in bar mm, U in kV).
%                        The IEC 60052 values are copyrighted by the IEC and
%                        are not distributed with this repository; see
%                        examples/iec60052/README.md.  Each overlay is guarded
%                        by \IfFileExists, so the figure builds without them
%                        and shows the computed curves alone.
%
% Column indices below are 0-based (\thisrowno) into sim_<D>cm.dat, whose
% layout for "incept1d pdiv MECH nodetachment.json --field sphere-sphere R
% --streamer-criterion 18 --write-to-file" is (1-based, see file header):
%    1  pd_bar_mm
%    5  U_kV[Baseline, sphere=positive]
%   15  U_kV[No detachment, sphere=positive]
%   28  U_kV[Streamer (C=18.0), Baseline]

\input{preamble.tex}
