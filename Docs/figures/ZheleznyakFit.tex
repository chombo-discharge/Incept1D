% NOTE: xmode/ymode must be given per axis, not inside a /.style (pgfplots
% re-evaluates them in TikZ context at \end{axis} and fails).
% Six-group two-stream reconstruction of the Zheleznyak absorption function.
%
% Data files (in the working directory when this is compiled), produced by
%   python3 Air/Zheleznyak.py --ngroups 8 --threshold 1E-6 --write-to-file zheleznyak
%   zheleznyak_absorption_curve.dat   col 0: p_O2 r (m Pa); 1: Zheleznyak;
%                                     2: six-group sum; 3-8: individual groups
%   zheleznyak_error_terms.dat        col 0: p_O2 r; 1: relative error (%)
\documentclass[10pt]{article}
\input{preamble.tex}
\usepackage[paperwidth=15cm,paperheight=14.8cm,margin=0in]{geometry}

\pgfplotsset{
  model/.style={solid, semithick, mark repeat=300,
    mark options={mark size=1.75pt, thin, solid, fill=white, fill opacity=1.0}},
  panel/.style={
    axis y line=left, axis x line=bottom, font=\small,
    width=\figwidth, height=\figheight, minor x tick num=4, grid=both,
    minor grid style={very thin, densely dotted}, axis line style={thick, black},
    xmax=1E3, cycle list name=exotic,
    xlabel={$p_{\text{O}_2}r$ (\si{\meter\pascal})},
  },
}
\setlength{\figwidth}{14.0cm}
\setlength{\figheight}{6.6cm}

\begin{document}
\centering
\begin{tikzpicture}
  \begin{axis}[panel, xmode=log, ymode=log, alias=absorption_curve, ymin=1E-9, ymax=10,
      title={a) Six-group Zheleznyak fit}, ylabel={Pressure-reduced absorption},
      boxedlegend, legend style={at={(0.05, 0.05)}, anchor=south west, legend columns=1}]
    \addplot+[model] table[x expr=\thisrowno{0}, y expr=\thisrowno{1}] {zheleznyak_absorption_curve.dat};
    \addplot+[model, densely dashed, mark repeat=200] table[x expr=\thisrowno{0}, y expr=\thisrowno{2}] {zheleznyak_absorption_curve.dat};
    \addplot[black, densely dashed] table[x expr=\thisrowno{0}, y expr=\thisrowno{3}] {zheleznyak_absorption_curve.dat};
    \addplot[black, densely dashed] table[x expr=\thisrowno{0}, y expr=\thisrowno{4}] {zheleznyak_absorption_curve.dat};
    \addplot[black, densely dashed] table[x expr=\thisrowno{0}, y expr=\thisrowno{5}] {zheleznyak_absorption_curve.dat};
    \addplot[black, densely dashed] table[x expr=\thisrowno{0}, y expr=\thisrowno{6}] {zheleznyak_absorption_curve.dat};
    \addplot[black, densely dashed] table[x expr=\thisrowno{0}, y expr=\thisrowno{7}] {zheleznyak_absorption_curve.dat};
    \addplot[black, densely dashed] table[x expr=\thisrowno{0}, y expr=\thisrowno{8}] {zheleznyak_absorption_curve.dat};
    \addlegendentry{Zheleznyak model};
    \addlegendentry{Six-group model};
    \addlegendentry{Individual groups};
  \end{axis}

  \begin{axis}[panel, xmode=log, at=(absorption_curve.south west), anchor=north west,
      yshift=-6em, ymax=1.0,
      title={b) Six-group error}, ylabel={Error (\si{\percent})}]
    \addplot+[model, mark repeat=50] table[x expr=\thisrowno{0}, y expr=\thisrowno{1}] {zheleznyak_error_terms.dat};
  \end{axis}
\end{tikzpicture}
\end{document}
